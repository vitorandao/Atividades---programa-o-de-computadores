numero = int(input("Digite um número de 4 dígitos: "))

# Validação de faixa
if numero < 0 or numero > 9999:
    print("Erro: número fora da faixa (0000 a 9999).")

else:

    while True:

        # Separar dígitos
        d1 = numero // 1000
        d2 = (numero % 1000) // 100
        d3 = (numero % 100) // 10
        d4 = numero % 10

        # Validação de repetição
        c1 = 1
        if d1 == d2: c1 += 1
        if d1 == d3: c1 += 1
        if d1 == d4: c1 += 1

        c2 = 1
        if d2 == d1: c2 += 1
        if d2 == d3: c2 += 1
        if d2 == d4: c2 += 1

        c3 = 1
        if d3 == d1: c3 += 1
        if d3 == d2: c3 += 1
        if d3 == d4: c3 += 1

        c4 = 1
        if d4 == d1: c4 += 1
        if d4 == d2: c4 += 1
        if d4 == d3: c4 += 1

        if c1 > 2 or c2 > 2 or c3 > 2 or c4 > 2:
            print("Erro: muitos dígitos repetidos.")
            break

        # Ordenação manual
        if d1 > d2:
            d1, d2 = d2, d1
        if d1 > d3:
            d1, d3 = d3, d1
        if d1 > d4:
            d1, d4 = d4, d1

        if d2 > d3:
            d2, d3 = d3, d2
        if d2 > d4:
            d2, d4 = d4, d2

        if d3 > d4:
            d3, d4 = d4, d3

        asc = d1*1000 + d2*100 + d3*10 + d4
        desc = d4*1000 + d3*100 + d2*10 + d1

        resultado = desc - asc

        print(desc, "-", asc, "=", resultado)

        if resultado == 6174:
            print("Constante de Kaprekar atingida!")
            break

        numero = resultado
